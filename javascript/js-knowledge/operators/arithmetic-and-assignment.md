```js
let x: number = 10;
x + 3;    // 13
x - 3;    // 7
x * 3;    // 30
x / 4;    // 2.5
x % 3;    // 1 (remainder)
x ** 2;   // 100 (exponentiation)
x++;      // 10 (returns old value, then increments to 11)
--x;      // 10 (decrements to 10, then returns)

x += 5;   // 15 (also -=, *=, /=, %=, **=)
```
